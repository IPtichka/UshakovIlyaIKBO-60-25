dst="/usr/local/bin/$(basename "$src")"

if [ ! -f "$src" ]; then
    echo "Файл не найден: $src" >&2
    exit 1
fi

if [ ! -w /usr/local/bin ]; then
    echo "Нужны права root. Запусти через sudo: sudo $0 $src" >&2
    exit 1
fi

chmod 755 "$src"

cp "$src" "$dst"

echo "Зарегистрировано: $dst"
